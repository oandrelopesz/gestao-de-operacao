# Aba Ideias — Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Uma aba `/ideias` que guarda ideias de implementação (neste sistema, em outro sistema ou para um cliente) em cards filtráveis, e um botão por card que cria a ideia como demanda na coluna do Kanban que o usuário escolher.

**Architecture:** Uma tabela `ideias` no Supabase, um módulo puro `src/lib/ideiasTexto.ts` (tipos, normalização de links, montagem da descrição da demanda, filtro), um serviço `src/lib/ideias.ts` (CRUD + `enviarIdeiaParaDemandas`), dois diálogos em `src/components/ideias/` e a página `src/routes/_authenticated/ideias.tsx` seguindo o padrão da aba Ver depois. O diálogo de envio busca `demanda_colunas` no banco toda vez que abre, então as colunas são sempre as da aba Demandas.

**Tech Stack:** React 19, TanStack Router (file routes), Supabase JS, shadcn/ui (Dialog, Select, Tabs, Card, Badge), lucide-react, sonner, `node --test` (Node 24 executa `.ts` direto).

**Spec:** `docs/superpowers/specs/2026-09-21-ideias-design.md`

/**
 * Regras puras da aba Ideias (sem Supabase nem alias "@/"), para rodar em `node --test`.
 */

export type TipoIdeia = "este_sistema" | "outro_sistema" | "cliente";

export const TIPOS_IDEIA: { id: TipoIdeia; label: string; hex: string }[] = [
  { id: "este_sistema", label: "Este sistema", hex: "#8b5cf6" },
  { id: "outro_sistema", label: "Outro sistema", hex: "#06b6d4" },
  { id: "cliente", label: "Para cliente", hex: "#f59e0b" },
];

export const tipoIdeiaMeta = (id: string) => TIPOS_IDEIA.find((t) => t.id === id) || TIPOS_IDEIA[0];

export function normalizarLink(link: string): string {
  const l = link.trim();
  if (!l) return "";
  return /^https?:\/\//i.test(l) ? l : `https://${l}`;
}

/** Normaliza cada link, descarta vazios e repetidos, mantendo a ordem digitada */
export function limparLinks(links: string[]): string[] {
  const out: string[] = [];
  for (const raw of links) {
    const l = normalizarLink(raw);
    if (l && !out.includes(l)) out.push(l);
  }
  return out;
}

export function dominioDoLink(link: string): string {
  try {
    return new URL(link).hostname.replace(/^www\./, "");
  } catch {
    return link;
  }
}

/** Texto gravado em demandas.descricao quando a ideia vira demanda */
export function montarDescricaoDemanda(ideia: { descricao: string | null; links: string[] }): string | null {
  const partes: string[] = [];
  const desc = (ideia.descricao || "").trim();
  if (desc) partes.push(desc);
  if (ideia.links.length > 0) partes.push(["Links de referência:", ...ideia.links.map((l) => `- ${l}`)].join("\n"));
  return partes.length > 0 ? partes.join("\n\n") : null;
}

export interface IdeiaFiltravel {
  tipo: string;
  agencia: string | null;
  titulo: string;
  descricao: string | null;
  links: string[];
}

/** Filtros da aba: tipo ("todos" = qualquer), agência ("todas" = qualquer) e busca livre */
export function filtrarIdeias<T extends IdeiaFiltravel>(ideias: T[], f: { tipo: string; agencia: string; busca: string }): T[] {
  const q = f.busca.trim().toLowerCase();
  return ideias.filter(
    (i) =>
      (f.tipo === "todos" || i.tipo === f.tipo) &&
      (f.agencia === "todas" || i.agencia === f.agencia) &&
      (!q ||
        i.titulo.toLowerCase().includes(q) ||
        (i.descricao || "").toLowerCase().includes(q) ||
        i.links.some((l) => l.toLowerCase().includes(q))),
  );
}

/** Coluna pré-selecionada ao enviar para demandas: o mesmo padrão do DemandaForm */
export function colunaPadrao(colunas: { id: string; sistema: boolean }[]): string {
  if (colunas.some((c) => c.id === "fazer_hoje")) return "fazer_hoje";
  return (colunas.find((c) => !c.sistema) || colunas[0])?.id || "";
}
```

import { supabase } from "@/integrations/supabase/client";
import { getUserId } from "@/lib/clientesService";
import { montarDescricaoDemanda, type TipoIdeia } from "@/lib/ideiasTexto";

export interface Ideia {
  id: string;
  tipo: TipoIdeia;
  titulo: string;
  descricao: string | null;
  links: string[];
  agencia: string | null;
  cliente_id: string | null;
  demanda_id: string | null;
  created_at: string;
  updated_at: string;
}

export type IdeiaInput = Pick<Ideia, "tipo" | "titulo" | "descricao" | "links" | "agencia" | "cliente_id">;

const ERRO_TABELA = "A tabela ideias ainda não existe. Rode supabase/migrations/20260921120000_ideias.sql no SQL Editor.";

export function mensagemErro(error: { message: string }): string {
  return /schema cache|does not exist|not find/i.test(error.message) ? ERRO_TABELA : error.message;
}

export async function fetchIdeias(): Promise<Ideia[]> {
  const { data, error } = await supabase.from("ideias").select("*").order("created_at", { ascending: false });
  if (error) throw error;
  return (data || []) as Ideia[];
}

export async function salvarIdeia(input: IdeiaInput, id?: string): Promise<void> {
  const { error } = id
    ? await supabase.from("ideias").update(input).eq("id", id)
    : await supabase.from("ideias").insert([input]);
  if (error) throw error;
}

export async function excluirIdeia(id: string): Promise<void> {
  const { error } = await supabase.from("ideias").delete().eq("id", id);
  if (error) throw error;
}

/**
 * Cria a demanda na coluna escolhida — título, descrição + links e cliente copiados da ideia,
 * no fim da coluna, como o "quick add" do Kanban — e grava demanda_id na ideia.
 */
export async function enviarIdeiaParaDemandas(ideia: Ideia, colunaId: string): Promise<string> {
  const userId = await getUserId();

  const { data: ultima, error: erroOrdem } = await supabase
    .from("demandas")
    .select("ordem")
    .eq("status", colunaId)
    .order("ordem", { ascending: false })
    .limit(1);
  if (erroOrdem) throw erroOrdem;
  const ordem = ultima && ultima.length > 0 ? (ultima[0].ordem || 0) + 1 : 0;

  const { data, error } = await supabase
    .from("demandas")
    .insert([{
      titulo: ideia.titulo,
      descricao: montarDescricaoDemanda(ideia),
      cliente_id: ideia.tipo === "cliente" ? ideia.cliente_id : null,
      status: colunaId,
      prioridade: "media",
      user_id: userId,
      ordem,
      recorrente: colunaId === "recorrentes",
    }])
    .select("id")
    .single();
  if (error) throw error;

  const { error: erroVinculo } = await supabase.from("ideias").update({ demanda_id: data.id }).eq("id", ideia.id);
  if (erroVinculo) throw erroVinculo;
  return data.id;
}
```

          </div>

                <Label htmlFor="ideia-agencia">Agência</Label>
                <Select value={agencia || undefined} onValueChange={(v) => { setAgencia(v); setClienteId(""); }}>
                  <SelectTrigger id="ideia-agencia"><SelectValue placeholder="Escolher" /></SelectTrigger>
                  <SelectContent>
                    {agencias.map((a) => (
                      <SelectItem key={a.id} value={a.id}>
                        <span className="inline-flex items-center gap-2">
                          <span className="h-2 w-2 rounded-full" style={{ backgroundColor: a.hex }} />
                          {a.label}
                        </span>
                      </SelectItem>
                    ))}
                  </SelectContent>
                </Select>
              </div>
              <div className="space-y-2">
                <Label htmlFor="ideia-cliente">Cliente</Label>
                <Select value={clienteId || undefined} onValueChange={setClienteId} disabled={!agencia}>
                  <SelectTrigger id="ideia-cliente"><SelectValue placeholder={agencia ? "Escolher" : "Escolha a agência"} /></SelectTrigger>

                    {clientesDaAgencia.map((c) => <SelectItem key={c.id} value={c.id}>{c.nome}</SelectItem>)}

### Task 5: Diálogo "Enviar para demandas"

**Files:**

- Regra: as colunas são buscadas no banco **toda vez que o diálogo abre** (nunca do cache), então colunas criadas/renomeadas/apagadas na aba Demandas aparecem aqui na hora.

```tsx
import { useEffect, useState } from "react";
import { Button } from "@/components/ui/button";
import { Label } from "@/components/ui/label";
import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from "@/components/ui/select";
import { Dialog, DialogContent, DialogDescription, DialogFooter, DialogHeader, DialogTitle } from "@/components/ui/dialog";
import { fetchDemandaColunas, type KanbanColumnDef } from "@/lib/demandaColunas";
import { enviarIdeiaParaDemandas, mensagemErro, type Ideia } from "@/lib/ideias";
import { colunaPadrao, montarDescricaoDemanda } from "@/lib/ideiasTexto";
import { clearCache } from "@/lib/pageCache";
import { CACHE_KEYS } from "@/lib/data";
import { toast } from "sonner";

interface Props {
  ideia: Ideia | null;
  onClose: () => void;
  onEnviada: (demandaId: string, coluna: KanbanColumnDef) => void;
}

export function EnviarParaDemandasDialog({ ideia, onClose, onEnviada }: Props) {
  const [colunas, setColunas] = useState<KanbanColumnDef[]>([]);
  const [carregando, setCarregando] = useState(false);
  const [colunaId, setColunaId] = useState("");
  const [enviando, setEnviando] = useState(false);

  // Busca as colunas do banco a cada abertura: a lista é sempre a da aba Demandas
  useEffect(() => {
    if (!ideia) return;
    let ativo = true;
    setCarregando(true);
    fetchDemandaColunas().then((cols) => {
      if (!ativo) return;
      setColunas(cols);
      setColunaId(colunaPadrao(cols));
      setCarregando(false);
    });
    return () => { ativo = false; };
  }, [ideia]);

  const enviar = async () => {
    if (!ideia || !colunaId) return;
    const coluna = colunas.find((c) => c.id === colunaId);
    if (!coluna) return;
    setEnviando(true);
    try {
      const demandaId = await enviarIdeiaParaDemandas(ideia, colunaId);
      // A aba Demandas recarrega do banco em vez de mostrar a lista antiga do cache
      clearCache(CACHE_KEYS.demandas);
      onEnviada(demandaId, coluna);
      onClose();
    } catch (e: any) {
      toast.error(mensagemErro(e));
    } finally {
      setEnviando(false);
    }
  };

  const previa = ideia ? montarDescricaoDemanda(ideia) : null;

  return (
    <Dialog open={!!ideia} onOpenChange={(o) => { if (!o) onClose(); }}>
      <DialogContent className="sm:max-w-[460px]">
        <DialogHeader>
          <DialogTitle>Enviar para demandas</DialogTitle>
          <DialogDescription>
            "{ideia?.titulo}" vira uma demanda na coluna escolhida, já com a descrição e os links.
            {ideia?.demanda_id && " Essa ideia já foi enviada antes; será criada outra demanda."}
          </DialogDescription>
        </DialogHeader>
        <div className="space-y-4">
          <div className="space-y-2">
            <Label htmlFor="ideia-coluna">Coluna</Label>
            <Select value={colunaId || undefined} onValueChange={setColunaId} disabled={carregando || colunas.length === 0}>
              <SelectTrigger id="ideia-coluna">
                <SelectValue placeholder={carregando ? "Carregando colunas..." : colunas.length === 0 ? "Nenhuma coluna no Kanban" : "Escolher"} />
              </SelectTrigger>
              <SelectContent>
                {colunas.map((c) => (
                  <SelectItem key={c.id} value={c.id}>
                    <span className="inline-flex items-center gap-2">
                      <span className="h-2 w-2 rounded-full" style={{ backgroundColor: c.hex }} />
                      {c.title}
                    </span>
                  </SelectItem>
                ))}
              </SelectContent>
            </Select>
          </div>
          {previa && (
            <div className="space-y-1">
              <Label>Descrição da demanda</Label>
              <pre className="whitespace-pre-wrap rounded-md border bg-muted/30 p-3 text-xs text-muted-foreground font-sans max-h-40 overflow-y-auto">{previa}</pre>
            </div>
          )}
        </div>
        <DialogFooter>
          <Button type="button" variant="ghost" onClick={onClose} disabled={enviando}>Cancelar</Button>
          <Button type="button" onClick={enviar} disabled={enviando || carregando || !colunaId}>
            {enviando ? "Enviando..." : "Criar demanda"}
          </Button>
        </DialogFooter>
      </DialogContent>
    </Dialog>
  );
}

                  {cliente && <p className="text-[11px] text-muted-foreground">Cliente: {cliente}</p>}
                  {ideia.descricao && <p className="text-xs text-muted-foreground line-clamp-3 whitespace-pre-line">{ideia.descricao}</p>}
                  {ideia.links.length > 0 && (
                    <div className="flex flex-wrap gap-1 pt-1">
                      {ideia.links.map((l) => (
                        <a
                          key={l}
                          href={l}
                          target="_blank"
                          rel="noopener noreferrer"
                          className="inline-flex items-center gap-1 rounded-full border px-2 py-0.5 text-[10px] text-muted-foreground hover:text-foreground hover:border-foreground/40 transition-colors"
                          title={l}
                        >
                          <ExternalLink size={10} /> {dominioDoLink(l)}
                        </a>
                      ))}
                    </div>
                  )}
                </div>
                <CardContent className="p-4 pt-2 mt-auto">
                  <div className="flex items-center gap-1.5">
                    <Button size="sm" className="flex-1 gap-1.5" onClick={() => setEnviando(ideia)}>
                      <SquareKanban size={14} /> Enviar para demandas
                    </Button>
                    <Button size="sm" variant="ghost" className="h-8 w-8 p-0" onClick={() => setDialog({ ideia })} title="Editar"><Pencil size={14} /></Button>
                    <Button size="sm" variant="ghost" className="h-8 w-8 p-0 text-destructive hover:text-destructive" onClick={() => setToDelete(ideia)} title="Remover"><Trash2 size={14} /></Button>
                  </div>
                </CardContent>
              </Card>
            );
          })}
        </div>
      )}

      <IdeiaDialog state={dialog} clientes={clientes} onClose={() => setDialog(null)} onSaved={load} />

      <EnviarParaDemandasDialog
        ideia={enviando}
        onClose={() => setEnviando(null)}
        onEnviada={(demandaId, coluna) => {
          setIdeias((prev) => prev.map((i) => (i.id === enviando?.id ? { ...i, demanda_id: demandaId } : i)));
          toast.success(`Demanda criada em "${coluna.title}"`, {
            action: { label: "Ver demandas", onClick: () => navigate({ to: "/" }) },
          });
        }}
      />

      <AlertDialog open={!!toDelete} onOpenChange={(o) => { if (!o) setToDelete(null); }}>
        <AlertDialogContent>
          <AlertDialogHeader>
            <AlertDialogTitle>Remover "{toDelete?.titulo}"?</AlertDialogTitle>
            <AlertDialogDescription>A ideia sai da lista. Demandas já criadas a partir dela continuam no Kanban.</AlertDialogDescription>
          </AlertDialogHeader>
          <AlertDialogFooter>
            <AlertDialogCancel>Cancelar</AlertDialogCancel>
            <AlertDialogAction onClick={excluir} className="bg-destructive hover:bg-destructive/90">Remover</AlertDialogAction>
          </AlertDialogFooter>
        </AlertDialogContent>
      </AlertDialog>
    </div>
  );
}
```

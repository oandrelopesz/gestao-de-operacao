# CRM Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Transformar a aba Prospecção num CRM com colunas configuráveis por tipo, tentativas de follow-up, pontos de contato da reunião, agenda com Meet/convite, Fila do dia, Planilha, ações em massa e Dashboard na mesma aba.

**Architecture:** Regras em libs puras testadas (`src/lib/crm*.ts`, rodam no `node --test`); uma camada de dados (`src/lib/crmDados.ts`) que fala com o Supabase e com o Google Agenda; um contexto React (`CrmContexto`) que concentra estado e ações; componentes de tela em `src/components/crm/` consumindo só o contexto. A tabela `prospeccao` é mantida e ganha campos; tabelas novas `crm_colunas` e `crm_eventos`.

**Tech Stack:** TanStack Start/Router, React 19, Supabase JS (Lovable Cloud), dnd-kit, Recharts, shadcn/ui, sonner, Google Calendar API v3 pelo navegador.

**Spec:** `docs/superpowers/specs/2026-09-26-crm-design.md`

## Global Constraints

- Nenhuma dependência nova no `package.json` (o `bun.lockb` não pode ser regenerado nesta máquina).
- Libs puras `src/lib/crm*.ts`: sem imports `@/`; imports relativos **com** extensão `.ts`; tipos importados com `import type` (o Node tira os tipos e quebraria com import de tipo sem `type`); sem `enum`. Testes em `src/lib/crm*.test.ts`, rodados com `npm test` (`node --test src/lib/*.test.ts`).
- Fuso: America/Sao_Paulo = UTC-3 fixo (sem horário de verão desde 2019). Toda conta de data do CRM passa por `src/lib/crmFormato.ts`.
- Agenda: sempre o calendário `"primary"` (contato@). Notificações 60/30/10 min. Título "Reunião — {nome}{ (empresa)} · André Lopes".
- Tabelas/colunas novas não estão em `src/integrations/supabase/types.ts` (gerado pelo Lovable): usar `supabase as any` na camada de dados.
- Textos da interface em português do Brasil, com acentos. Cor nunca é o único sinal (sempre texto junto).
- Nunca `window.confirm`/`alert`: usar `AlertDialog`.
- Nada muda visualmente fora da aba CRM (exceto o menu: "Prospecção" → "CRM"; e a aba Financeiro usa o `FonteDialog` extraído sem mudar comportamento).
- Commits em português, terminando com `Co-Authored-By: Claude Opus 5.5 (1M context) <noreply@anthropic.com>`. Não commitar `CRM-Evoluir-Kids-Funcionalidades.md` nem `evoluir-kids-crm-codigo.zip`.
- A migration precisa estar aplicada (editor SQL do Lovable) antes de publicar.
- `routeTree.gen.ts` é regenerado pelo plugin do router ao rodar `npx vite build` ou o dev server; commitar a versão regenerada.

// Datas e números do CRM. Fuso de São Paulo = UTC-3 fixo (sem horário de verão desde 2019), então as contas
// não dependem do fuso da máquina. Sem dependências (roda no `node --test`).

const OFFSET_MS = 3 * 60 * 60 * 1000;
export const HORA_MS = 3_600_000;
export const DIA_MS = 86_400_000;
export const DIAS_SEMANA = ["domingo", "segunda", "terça", "quarta", "quinta", "sexta", "sábado"];
const DIAS_CURTOS = ["dom", "seg", "ter", "qua", "qui", "sex", "sáb"];

export interface PartesSP {
  ano: number;
  /** 1–12 */
  mes: number;
  dia: number;
  hora: number;
  minuto: number;
  /** 0 = domingo */
  diaSemana: number;
}

export function partesSP(d: Date): PartesSP {
  const l = new Date(d.getTime() - OFFSET_MS);
  return {
    ano: l.getUTCFullYear(), mes: l.getUTCMonth() + 1, dia: l.getUTCDate(),
    hora: l.getUTCHours(), minuto: l.getUTCMinutes(), diaSemana: l.getUTCDay(),
  };
}

/** Instante de uma data/hora de parede em SP (mes 1–12; o dia pode transbordar: 0 = último do mês anterior). */
export function deSP(ano: number, mes: number, dia: number, hora = 0, minuto = 0): Date {
  return new Date(Date.UTC(ano, mes - 1, dia, hora, minuto) + OFFSET_MS);
}

/** O dia de `d` em SP deslocado `dias`, às hh:mm. */
export function diaSPAs(d: Date, dias: number, hora: number, minuto = 0): Date {
  const p = partesSP(d);
  return deSP(p.ano, p.mes, p.dia + dias, hora, minuto);
}

export function mesmoDiaSP(a: Date, b: Date): boolean {
  const x = partesSP(a);
  const y = partesSP(b);
  return x.ano === y.ano && x.mes === y.mes && x.dia === y.dia;
}

const dois = (n: number) => String(n).padStart(2, "0");
const paraData = (v: string | Date) => (typeof v === "string" ? new Date(v) : v);

/** "26/09 às 14:30" */
export function formatarDataHora(v: string | Date): string {
  const p = partesSP(paraData(v));
  return `${dois(p.dia)}/${dois(p.mes)} às ${dois(p.hora)}:${dois(p.minuto)}`;
}

/** "qua 11:00" */
export function formatarDiaCurtoHora(v: string | Date): string {
  const p = partesSP(paraData(v));
  return `${DIAS_CURTOS[p.diaSemana]} ${dois(p.hora)}:${dois(p.minuto)}`;
}

/** "14:30" */
export function formatarHora(v: string | Date): string {
  const p = partesSP(paraData(v));
  return `${dois(p.hora)}:${dois(p.minuto)}`;
}

/** "26/09/2026" */
export function formatarData(v: string | Date): string {
  const p = partesSP(paraData(v));
  return `${dois(p.dia)}/${dois(p.mes)}/${p.ano}`;
}

/** Campos `<input type=date>` (aaaa-mm-dd) e `<input type=time>` (hh:mm) em SP → ISO. */
export function isoDeCampos(data: string, hora: string): string | null {
  const d = /^(\d{4})-(\d{2})-(\d{2})$/.exec(data);
  const h = /^(\d{2}):(\d{2})$/.exec(hora);
  if (!d || !h) return null;
  return deSP(Number(d[1]), Number(d[2]), Number(d[3]), Number(h[1]), Number(h[2])).toISOString();
}

/** ISO → campos de formulário em SP. */
export function camposDeIso(iso: string | null | undefined): { data: string; hora: string } {
  if (!iso) return { data: "", hora: "" };
  const p = partesSP(new Date(iso));
  return { data: `${p.ano}-${dois(p.mes)}-${dois(p.dia)}`, hora: `${dois(p.hora)}:${dois(p.minuto)}` };
}

const umaCasa = (n: number) => (Math.round(n * 10) / 10).toString().replace(".", ",");

/** "45 min", "3,5 h", "2,1 dias" */
export function formatarDuracao(horas: number | null): string {
  if (horas === null || !Number.isFinite(horas)) return "—";
  if (horas < 1) return `${Math.round(horas * 60)} min`;
  if (horas < 48) return `${umaCasa(horas)} h`;
  return `${umaCasa(horas / 24)} dias`;
}

const REAIS = new Intl.NumberFormat("pt-BR", { style: "currency", currency: "BRL" });
export function formatarReais(v: number | null | undefined): string {
  return REAIS.format(Number(v) || 0).replace(/ /g, " ");
}

/** 0.256 → "25,6%" */
export function formatarPct(v: number | null): string {
  if (v === null || !Number.isFinite(v)) return "—";
  return `${umaCasa(v * 100)}%`;
}

// Os 5 pontos de contato antes da reunião. Horários variam pelo período da call (spec §6). Sem dependências.
import type { LeadCrm, PontosContatoEstado } from "./crmTipos.ts";
import { diaSPAs, partesSP } from "./crmFormato.ts";

export type PeriodoCall = "manha" | "inicio_tarde" | "fim_dia" | "noite";
export const ORDEM_PERIODOS: PeriodoCall[] = ["manha", "inicio_tarde", "fim_dia", "noite"];
export const ROTULO_PERIODO: Record<PeriodoCall, string> = {
  manha: "Manhã",
  inicio_tarde: "Início da tarde",
  fim_dia: "Fim do dia",
  noite: "Noite",
};
export const NOMES_PONTOS = [
  "Aquecimento (3 dias antes)",
  "Confirmação na véspera",
  "Lembrete do dia",
  "Pré-call",
  "Link do Meet",
];

export function periodoDaCall(inicio: Date): PeriodoCall {
  const h = partesSP(inicio).hora;
  if (h < 12) return "manha";
  if (h < 15) return "inicio_tarde";
  if (h < 18) return "fim_dia";
  return "noite";
}

/** Horário dos pontos 1 e 2 (D-3 e D-1) em cada período */
const BASE: Record<PeriodoCall, [number, number]> = {
  manha: [10, 0],
  inicio_tarde: [11, 0],
  fim_dia: [15, 0],
  noite: [18, 30],
};
const MIN = 60_000;

/** Os 5 horários para uma reunião que começa em `inicio`. */
export function horariosDosPontos(inicio: Date): Date[] {
  const periodo = periodoDaCall(inicio);
  const [h, m] = BASE[periodo];
  const t = inicio.getTime();
  const p1 = diaSPAs(inicio, -3, h, m);
  const p2 = diaSPAs(inicio, -1, h, m);
  const p3 =
    periodo === "manha" ? diaSPAs(inicio, -1, 19, 0)
    : periodo === "inicio_tarde" ? diaSPAs(inicio, 0, 8, 30)
    : periodo === "fim_dia" ? diaSPAs(inicio, 0, 9, 0)
    : diaSPAs(inicio, 0, 12, 0);
  let p4: Date;
  if (periodo === "manha") {
    const umaHoraAntes = Math.max(t - 60 * MIN, diaSPAs(inicio, 0, 7, 30).getTime());
    p4 = new Date(Math.min(umaHoraAntes, t - 20 * MIN));
  } else {
    p4 = new Date(t - 120 * MIN);
  }
  const p5 = new Date(t - 10 * MIN);
  return [p1, p2, p3, p4, p5];
}

export type EstadoPonto = "feito" | "nao_se_aplica" | "atrasado" | "pendente";

export interface PontoCalculado {
  n: number;
  nome: string;
  quando: Date;
  estado: EstadoPonto;
  feitoEm: string | null;
}

type LeadPontos = Pick<LeadCrm, "reuniao_em" | "reuniao_marcada_em" | "pontos_contato">;

const mesmoInstante = (a: string, b: string) => new Date(a).getTime() === new Date(b).getTime();

/** Feitos só valem para a reunião atual (`ref` = reuniao_em). */
function estadoAtual(lead: LeadPontos): PontosContatoEstado {
  const e = lead.pontos_contato ?? {};
  if (!lead.reuniao_em) return {};
  if (e.ref && !mesmoInstante(e.ref, lead.reuniao_em)) return { marcada_em: lead.reuniao_marcada_em ?? undefined };
  return e;
}

export function calcularPontos(lead: LeadPontos, agora: Date): PontoCalculado[] {
  if (!lead.reuniao_em) return [];
  const estado = estadoAtual(lead);
  const marcadaIso = estado.marcada_em ?? lead.reuniao_marcada_em;
  const marcada = marcadaIso ? new Date(marcadaIso).getTime() : null;
  const feitos = estado.feitos ?? {};
  return horariosDosPontos(new Date(lead.reuniao_em)).map((quando, i) => {
    const n = i + 1;
    const feitoEm = feitos[String(n)] ?? null;
    let e: EstadoPonto;
    if (feitoEm) e = "feito";
    else if (marcada !== null && quando.getTime() < marcada) e = "nao_se_aplica";
    else if (quando.getTime() <= agora.getTime()) e = "atrasado";
    else e = "pendente";
    return { n, nome: NOMES_PONTOS[i], quando, estado: e, feitoEm };
  });
}

export interface ResumoPontos {
  feitos: number;
  aplicaveis: number;
  proximo: PontoCalculado | null;
  /** O atrasado mais recente */
  atrasado: PontoCalculado | null;
}

export function resumoPontos(pontos: PontoCalculado[]): ResumoPontos {
  const atrasados = pontos.filter((p) => p.estado === "atrasado");
  return {
    feitos: pontos.filter((p) => p.estado === "feito").length,
    aplicaveis: pontos.filter((p) => p.estado !== "nao_se_aplica").length,
    proximo: pontos.find((p) => p.estado === "pendente") ?? null,
    atrasado: atrasados.length ? atrasados[atrasados.length - 1] : null,
  };
}

/** Reunião nova ou com data nova: pontos recomeçam. */
export function reiniciarPontos(reuniaoEm: string, agora: Date): PontosContatoEstado {
  return { ref: new Date(reuniaoEm).toISOString(), marcada_em: agora.toISOString(), feitos: {} };
}

export function marcarPonto(lead: LeadPontos, n: number, feito: boolean, agora: Date): PontosContatoEstado {
  if (!lead.reuniao_em) throw new Error("Lead sem reunião marcada");
  const atual = estadoAtual(lead);
  const base: PontosContatoEstado = {
    ref: new Date(lead.reuniao_em).toISOString(),
    marcada_em: atual.marcada_em ?? lead.reuniao_marcada_em ?? agora.toISOString(),
    feitos: { ...(atual.feitos ?? {}) },
  };
  if (feito) base.feitos![String(n)] = agora.toISOString();
  else delete base.feitos![String(n)];
  return base;
}

// Tentativas de follow-up (prospecção e proposta): teto por coluna, cadência do próximo contato. Sem dependências.
import type { ColunaCrm, LeadCrm, NovoEvento, TipoColuna } from "./crmTipos.ts";
import { tetoDa } from "./crmColunas.ts";
import { diaSPAs } from "./crmFormato.ts";

/** Dias até o próximo contato depois da tentativa N (índice N-1), sempre às 10h. */
export const CADENCIA_DIAS = [1, 2, 2, 3, 3, 4];

export type EscopoTentativa = "prospeccao" | "proposta";

export function escopoDoTipo(tipo: TipoColuna): EscopoTentativa | null {
  if (tipo === "follow_up") return "prospeccao";
  if (tipo === "follow_up_proposta") return "proposta";
  return null;
}

export function ordinal(n: number): string {
  return `${n}ª`;
}

export function proximoContatoApos(n: number, agora: Date): string | null {
  const dias = CADENCIA_DIAS[n - 1];
  return dias === undefined ? null : diaSPAs(agora, dias, 10, 0).toISOString();
}

const feitasNo = (lead: LeadCrm, escopo: EscopoTentativa) =>
  (escopo === "prospeccao" ? lead.follow_up : lead.follow_up_proposta) ?? 0;

const nomeEscopo = (escopo: EscopoTentativa) => (escopo === "prospeccao" ? "follow-up" : "follow-up de proposta");

export interface ResultadoTentativa {
  patch: Partial<LeadCrm>;
  eventos: NovoEvento[];
}

/** Próxima tentativa do escopo da coluna. Devolve a mensagem de erro quando não dá. */
export function registrarTentativa(lead: LeadCrm, coluna: ColunaCrm, agora: Date): ResultadoTentativa | string {
  const escopo = escopoDoTipo(coluna.tipo);
  if (!escopo) return "Esta coluna não conta tentativas.";
  const teto = tetoDa(coluna);
  const atual = feitasNo(lead, escopo);
  if (atual >= teto) return `As ${teto} tentativas acabaram.`;
  const n = atual + 1;
  const patch: Partial<LeadCrm> = escopo === "prospeccao" ? { follow_up: n } : { follow_up_proposta: n };
  patch.proximo_contato_em = n >= teto ? null : proximoContatoApos(n, agora);
  if (!lead.primeiro_contato_em) patch.primeiro_contato_em = agora.toISOString();
  return {
    patch,
    eventos: [{ tipo: "tentativa", descricao: `${ordinal(n)} tentativa de ${nomeEscopo(escopo)}`, dados: { n, escopo } }],
  };
}

/** Entrar numa coluna de follow-up conta a 1ª tentativa se o contador ainda está em zero. */
export function entrarEmFollowUp(lead: LeadCrm, coluna: ColunaCrm, agora: Date): ResultadoTentativa | null {
  const escopo = escopoDoTipo(coluna.tipo);
  if (!escopo || feitasNo(lead, escopo) > 0) return null;
  const r = registrarTentativa(lead, coluna, agora);
  if (typeof r === "string") return null;
  r.eventos[0] = {
    ...r.eventos[0],
    descricao: `${r.eventos[0].descricao} (ao entrar na coluna)`,
    dados: { ...r.eventos[0].dados, ao_entrar: true },
  };
  return r;
}

export interface InfoTentativas {
  escopo: EscopoTentativa;
  feitas: number;
  teto: number;
  noLimite: boolean;
  rotuloBotao: string;
}

export function infoTentativas(lead: LeadCrm, coluna: ColunaCrm): InfoTentativas | null {
  const escopo = escopoDoTipo(coluna.tipo);
  if (!escopo) return null;
  const feitas = feitasNo(lead, escopo);
  const teto = tetoDa(coluna);
  const noLimite = feitas >= teto;
  return { escopo, feitas, teto, noLimite, rotuloBotao: noLimite ? "Limite atingido" : `Registrar ${ordinal(feitas + 1)} tentativa` };
}

/** Fora da coluna de follow-up: "Respondeu na 3ª tentativa de follow-up" / "Parou na 5ª ...". */
export function textoCongelado(lead: LeadCrm, tipoAtual: TipoColuna): string | null {
  if (escopoDoTipo(tipoAtual)) return null;
  const n = lead.follow_up ?? 0;
  if (n <= 0) return null;
  const parou = tipoAtual === "perdido" || tipoAtual === "desqualificado";
  return `${parou ? "Parou" : "Respondeu"} na ${ordinal(n)} tentativa de follow-up`;
}

// O que acontece ao mover um lead: patch, eventos do histórico e a ação da agenda (spec §4). Sem dependências.
import type { ColunaCrm, LeadCrm, MotivoRemarcacao, NovoEvento } from "./crmTipos.ts";
import { emailValido, exigenciaDe, MOTIVOS_REMARCACAO } from "./crmColunas.ts";
import { entrarEmFollowUp } from "./crmTentativas.ts";
import { calcularPontos, periodoDaCall, reiniciarPontos, resumoPontos } from "./crmPontos.ts";
import { formatarDataHora, formatarReais, partesSP } from "./crmFormato.ts";

export interface DadosReuniao {
  em: string;
  duracaoMin: number;
  /** undefined = não mexe no e-mail do lead; null/"" = apaga */
  email?: string | null;
}

export interface DadosJanela {
  reuniao?: DadosReuniao;
  remarcacao?: { motivo: MotivoRemarcacao; nova?: DadosReuniao | null };
  valorProposta?: number | null;
  valorFechamento?: number;
  motivo?: string;
}

export type AcaoAgenda = "criar" | "mover" | "convidar" | "cancelar" | null;

export interface PlanoMovimento {
  patch: Partial<LeadCrm>;
  eventos: NovoEvento[];
  agenda: AcaoAgenda;
}

export function validarReuniao(r?: DadosReuniao | null): string | null {
  if (!r || !r.em || Number.isNaN(new Date(r.em).getTime())) return "Informe a data e a hora da reunião.";
  if (!(r.duracaoMin >= 5 && r.duracaoMin <= 480)) return "Duração inválida.";
  if (r.email && !emailValido(r.email)) return "E-mail inválido.";
  return null;
}

export function validarMovimento(lead: Pick<LeadCrm, "reuniao_em">, destino: ColunaCrm, dados: DadosJanela): string | null {
  switch (exigenciaDe(destino.tipo, lead)) {
    case "reuniao":
      return validarReuniao(dados.reuniao);
    case "remarcacao": {
      const r = dados.remarcacao;
      if (!r) return "Escolha o que aconteceu.";
      if (r.motivo === "pediu_remarcar" && !r.nova) return "Informe a nova data e hora.";
      return r.nova ? validarReuniao(r.nova) : null;
    }
    case "ganho":
      return dados.valorFechamento && dados.valorFechamento > 0 ? null : "Informe o valor fechado (maior que zero).";
    case "motivo":
      return dados.motivo?.trim() ? null : "Escolha o motivo.";
    case "proposta":
      return dados.valorProposta != null && dados.valorProposta < 0 ? "Valor inválido." : null;
    default:
      return null;
  }
}

const mesmoInstante = (a: string | null, b: string | null) =>
  !!a && !!b && new Date(a).getTime() === new Date(b).getTime();

/** Reunião nova ou alterada: campos, marcos, pontos e o que fazer na agenda. */
export function planejarReuniao(lead: LeadCrm, r: DadosReuniao, agora: Date): PlanoMovimento {
  const agoraIso = agora.toISOString();
  const emIso = new Date(r.em).toISOString();
  const mudouData = !mesmoInstante(lead.reuniao_em, emIso);
  const mudouDuracao = (lead.reuniao_duracao_min ?? 30) !== r.duracaoMin;
  const emailNovo = r.email === undefined ? lead.email : r.email?.trim() || null;
  const mudouEmail = (emailNovo ?? null) !== (lead.email ?? null);

  const patch: Partial<LeadCrm> = { reuniao_em: emIso, reuniao_duracao_min: r.duracaoMin };
  if (mudouEmail) patch.email = emailNovo;
  if (mudouData) {
    patch.reuniao_marcada_em = agoraIso;
    patch.pontos_contato = reiniciarPontos(emIso, agora);
  }
  if (!lead.primeira_reuniao_marcada_em) {
    patch.primeira_reuniao_marcada_em = agoraIso;
    patch.follow_ups_ate_reuniao = lead.follow_up ?? 0;
  }

  const eventos: NovoEvento[] = [];
  if (mudouData || mudouDuracao) {
    const verbo = lead.reuniao_em && mudouData ? "Reunião movida para" : "Reunião marcada para";
    eventos.push({
      tipo: "reuniao",
      descricao: `${verbo} ${formatarDataHora(emIso)} (${r.duracaoMin} min)`,
      dados: { em: emIso, duracao_min: r.duracaoMin, anterior: lead.reuniao_em },
    });
  }

  let agenda: AcaoAgenda = null;
  if (!lead.evento_google_id) agenda = "criar";
  else if (mudouData || mudouDuracao) agenda = "mover";
  else if (mudouEmail) agenda = "convidar";
  return { patch, eventos, agenda };
}

function resultadoReuniao(lead: LeadCrm, compareceu: boolean, agora: Date): NovoEvento {
  const em = lead.reuniao_em;
  if (!em) {
    return { tipo: "reuniao_resultado", descricao: compareceu ? "Compareceu à reunião" : "Não compareceu à reunião", dados: { compareceu, reuniao_em: null } };
  }
  const inicio = new Date(em);
  const r = resumoPontos(calcularPontos(lead, agora));
  return {
    tipo: "reuniao_resultado",
    descricao: `${compareceu ? "Compareceu" : "Não compareceu"} à reunião de ${formatarDataHora(em)}`,
    dados: {
      compareceu,
      reuniao_em: em,
      periodo: periodoDaCall(inicio),
      dia_semana: partesSP(inicio).diaSemana,
      pontos_feitos: r.feitos,
      pontos_aplicaveis: r.aplicaveis,
    },
  };
}

const rotuloRemarcacao = (m: MotivoRemarcacao) => MOTIVOS_REMARCACAO.find((x) => x.id === m)?.rotulo ?? m;

/** Plano completo de um movimento de coluna. Assume `validarMovimento` já conferido. */
export function planejarMovimento(
  lead: LeadCrm,
  origem: ColunaCrm | null,
  destino: ColunaCrm,
  dados: DadosJanela,
  agora: Date,
): PlanoMovimento {
  const agoraIso = agora.toISOString();
  let patch: Partial<LeadCrm> = { coluna_id: destino.id, coluna_desde: agoraIso };
  const eventos: NovoEvento[] = [{
    tipo: "mudanca_coluna",
    descricao: `${origem?.titulo ?? "—"} → ${destino.titulo}`,
    dados: { de: origem?.id ?? null, para: destino.id, tipo: destino.tipo },
  }];
  let agenda: AcaoAgenda = null;
  const encerra = destino.tipo === "perdido" || destino.tipo === "desqualificado";
  if (!lead.primeiro_contato_em && !encerra) patch.primeiro_contato_em = agoraIso;

  const atual = (): LeadCrm => ({ ...lead, ...patch });
  const juntar = (p: PlanoMovimento) => {
    patch = { ...patch, ...p.patch };
    eventos.push(...p.eventos);
    if (p.agenda) agenda = p.agenda;
  };

  switch (destino.tipo) {
    case "follow_up":
    case "follow_up_proposta": {
      const r = entrarEmFollowUp(atual(), destino, agora);
      if (r) juntar({ patch: r.patch, eventos: r.eventos, agenda: null });
      break;
    }
    case "reuniao":
      if (dados.reuniao) juntar(planejarReuniao(atual(), dados.reuniao, agora));
      break;
    case "confirmacao":
      if (!lead.reuniao_em && dados.reuniao) juntar(planejarReuniao(atual(), dados.reuniao, agora));
      break;
    case "remarcacao": {
      const r = dados.remarcacao;
      if (!r) break;
      patch.motivo_remarcacao = r.motivo;
      eventos.push({ tipo: "remarcacao", descricao: `Remarcação: ${rotuloRemarcacao(r.motivo)}`, dados: { motivo: r.motivo } });
      if (r.motivo === "no_show") {
        patch.no_shows = (lead.no_shows ?? 0) + 1;
        eventos.push(resultadoReuniao(lead, false, agora));
        if (!r.nova) {
          patch.reuniao_em = null;
          patch.pontos_contato = {};
        }
      }
      if (r.nova) {
        patch.remarcacoes = (lead.remarcacoes ?? 0) + 1;
        juntar(planejarReuniao(atual(), r.nova, agora));
      }
      break;
    }
    case "realizada":
      patch.compareceu = true;
      if (!lead.primeira_reuniao_realizada_em) patch.primeira_reuniao_realizada_em = agoraIso;
      if (lead.reuniao_em) eventos.push(resultadoReuniao(lead, true, agora));
      break;
    case "proposta":
      if (dados.valorProposta != null) patch.valor_proposta = dados.valorProposta;
      if (!lead.proposta_enviada_em) patch.proposta_enviada_em = agoraIso;
      break;
    case "ganho": {
      const valor = dados.valorFechamento ?? 0;
      patch.valor_fechamento = valor;
      if (!lead.fechado_em) {
        patch.fechado_em = agoraIso;
        patch.follow_ups_ate_fechamento = lead.follow_up_proposta ?? 0;
      }
      eventos.push({ tipo: "ganho", descricao: `Fechado: ${formatarReais(valor)}`, dados: { valor } });
      break;
    }
    case "perdido":
    case "desqualificado": {
      const motivo = (dados.motivo ?? "").trim();
      if (destino.tipo === "perdido") {
        patch.motivo_perda = motivo;
        patch.perdido_em = agoraIso;
        eventos.push({ tipo: "perda", descricao: `Perdido: ${motivo}`, dados: { motivo } });
      } else {
        patch.motivo_desqualificacao = motivo;
        patch.desqualificado_em = agoraIso;
        eventos.push({ tipo: "desqualificacao", descricao: `Desqualificado: ${motivo}`, dados: { motivo } });
      }
      if (lead.reuniao_em && new Date(lead.reuniao_em).getTime() > agora.getTime()) {
        patch.reuniao_em = null;
        patch.pontos_contato = {};
        if (lead.evento_google_id) agenda = "cancelar";
      }
      break;
    }
    default:
      break;
  }
  return { patch, eventos, agenda };
}

}

  if (d.length === 10) return `(${d.slice(0, 2)}) ${d.slice(2, 6)}-${d.slice(6)}`;
  return bruto;

// Período (atalhos em SP), critério (entrada × movimentação), busca e etiqueta. Sem dependências.
import type { EventoCrm, LeadCrm, Trafego } from "./crmTipos.ts";
import { deSP, diaSPAs, partesSP } from "./crmFormato.ts";

export type AtalhoPeriodo = "hoje" | "ontem" | "7d" | "30d" | "mes" | "mes_passado" | "maximo" | "personalizado";
export type CriterioPeriodo = "entrada" | "movimentacao";

export interface FiltroPeriodo {
  atalho: AtalhoPeriodo;
  /** aaaa-mm-dd (personalizado) */
  de?: string | null;
  ate?: string | null;
  criterio: CriterioPeriodo;
}

export interface FiltroCrm {
  periodo: FiltroPeriodo;
  busca: string;
  trafego: "todos" | Trafego;
}

export const ATALHOS_PERIODO: { id: AtalhoPeriodo; rotulo: string }[] = [
  { id: "hoje", rotulo: "Hoje" },
  { id: "ontem", rotulo: "Ontem" },
  { id: "7d", rotulo: "Últimos 7 dias" },
  { id: "30d", rotulo: "Últimos 30 dias" },
  { id: "mes", rotulo: "Este mês" },
  { id: "mes_passado", rotulo: "Mês passado" },
  { id: "maximo", rotulo: "Período máximo" },
  { id: "personalizado", rotulo: "Personalizado" },
];

export const FILTRO_CRM_PADRAO: FiltroCrm = {
  periodo: { atalho: "maximo", criterio: "entrada" },
  busca: "",
  trafego: "todos",
};

export interface Intervalo {
  inicio: Date | null;
  fim: Date | null;
}

const doCampo = (s: string | null | undefined, somaDias = 0): Date | null => {
  const m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(s ?? "");
  return m ? deSP(Number(m[1]), Number(m[2]), Number(m[3]) + somaDias) : null;
};

export function intervaloDo(f: FiltroPeriodo, agora: Date): Intervalo {
  const p = partesSP(agora);
  switch (f.atalho) {
    case "hoje":
      return { inicio: diaSPAs(agora, 0, 0), fim: diaSPAs(agora, 1, 0) };
    case "ontem":
      return { inicio: diaSPAs(agora, -1, 0), fim: diaSPAs(agora, 0, 0) };
    case "7d":
      return { inicio: diaSPAs(agora, -6, 0), fim: diaSPAs(agora, 1, 0) };
    case "30d":
      return { inicio: diaSPAs(agora, -29, 0), fim: diaSPAs(agora, 1, 0) };
    case "mes":
      return { inicio: deSP(p.ano, p.mes, 1), fim: deSP(p.ano, p.mes + 1, 1) };
    case "mes_passado":
      return { inicio: deSP(p.ano, p.mes - 1, 1), fim: deSP(p.ano, p.mes, 1) };
    case "personalizado":
      return { inicio: doCampo(f.de), fim: doCampo(f.ate, 1) };
    default:
      return { inicio: null, fim: null };
  }
}

export function dentro(iso: string | null | undefined, i: Intervalo): boolean {
  if (!iso) return false;
  const t = new Date(iso).getTime();
  return (!i.inicio || t >= i.inicio.getTime()) && (!i.fim || t < i.fim.getTime());
}

export function normalizarBusca(s: string): string {
  return s.normalize("NFD").replace(/[̀-ͯ]/g, "").toLowerCase().trim();
}

export function agruparEventos(eventos: EventoCrm[]): Map<string, EventoCrm[]> {
  const m = new Map<string, EventoCrm[]>();
  for (const e of eventos) {
    const lista = m.get(e.lead_id);
    if (lista) lista.push(e);
    else m.set(e.lead_id, [e]);
  }
  return m;
}

export function filtrarLeads(leads: LeadCrm[], eventosPorLead: Map<string, EventoCrm[]>, f: FiltroCrm, agora: Date): LeadCrm[] {
  const i = intervaloDo(f.periodo, agora);
  const semLimite = !i.inicio && !i.fim;
  const termo = normalizarBusca(f.busca);
  const termoDigitos = f.busca.replace(/\D/g, "");
  return leads.filter((l) => {
    if (f.trafego !== "todos" && l.trafego !== f.trafego) return false;
    if (!semLimite) {
      const ok = f.periodo.criterio === "entrada"
        ? dentro(l.created_at, i)
        : (eventosPorLead.get(l.id) ?? []).some((e) => dentro(e.created_at, i));
      if (!ok) return false;
    }
    if (termo) {
      const texto = normalizarBusca([l.nome, l.empresa, l.nicho, l.email, l.cidade].filter(Boolean).join(" "));
      const noTelefone = termoDigitos.length >= 4 && (l.telefone ?? "").replace(/\D/g, "").includes(termoDigitos);
      if (!texto.includes(termo) && !noTelefone) return false;
    }
    return true;
  });
}

// Urgência do card e blocos da Fila do dia. Sem dependências.
import type { ColunaCrm, LeadCrm } from "./crmTipos.ts";
import { colunaInicial, TIPOS_COM_PONTOS, TIPOS_FINAIS } from "./crmColunas.ts";
import { calcularPontos, resumoPontos } from "./crmPontos.ts";
import { formatarDataHora, formatarHora, HORA_MS, mesmoDiaSP } from "./crmFormato.ts";

export type Urgencia = "sem_contato" | "contato_vencido" | "reuniao_proxima" | "em_dia";

export interface InfoUrgencia {
  tipo: Urgencia;
  texto: string;
}

const tipoDaColuna = (colunas: ColunaCrm[], id: string | null) => colunas.find((c) => c.id === id)?.tipo ?? null;

export function urgenciaDoLead(lead: LeadCrm, colunas: ColunaCrm[], agora: Date): InfoUrgencia {
  const tipo = tipoDaColuna(colunas, lead.coluna_id);
  if (tipo && TIPOS_FINAIS.includes(tipo)) return { tipo: "em_dia", texto: "" };
  const inicial = colunaInicial(colunas);
  const naInicial = !lead.coluna_id || lead.coluna_id === inicial?.id;
  if (naInicial && (lead.follow_up ?? 0) === 0 && agora.getTime() - new Date(lead.created_at).getTime() >= 24 * HORA_MS) {
    return { tipo: "sem_contato", texto: "Sem contato há mais de 24 h" };
  }
  if (lead.proximo_contato_em && new Date(lead.proximo_contato_em).getTime() <= agora.getTime()) {
    return { tipo: "contato_vencido", texto: `Contato vencido (${formatarDataHora(lead.proximo_contato_em)})` };
  }
  if (lead.reuniao_em) {
    const falta = new Date(lead.reuniao_em).getTime() - agora.getTime();
    if (falta > 0 && falta <= 24 * HORA_MS) return { tipo: "reuniao_proxima", texto: `Reunião em menos de 24 h (${formatarDataHora(lead.reuniao_em)})` };
  }
  return { tipo: "em_dia", texto: "" };
}

export type BlocoFila = "atrasados" | "pontos_atrasados" | "novos" | "hoje" | "reunioes_hoje" | "reunioes_amanha";

export interface ItemFila {
  lead: LeadCrm;
  motivo: string;
  /** Para ordenar dentro do bloco */
  chave: number;
}

export interface GrupoFila {
  bloco: BlocoFila;
  titulo: string;
  itens: ItemFila[];
}

const TITULOS: Record<BlocoFila, string> = {
  atrasados: "Contatos atrasados",
  pontos_atrasados: "Pontos de contato atrasados",
  novos: "Novos sem contato",
  hoje: "Para hoje",
  reunioes_hoje: "Reuniões de hoje",
  reunioes_amanha: "Reuniões de amanhã",
};
const ORDEM: BlocoFila[] = ["atrasados", "pontos_atrasados", "novos", "hoje", "reunioes_hoje", "reunioes_amanha"];

export function montarFila(leads: LeadCrm[], colunas: ColunaCrm[], agora: Date): GrupoFila[] {
  const grupos = new Map<BlocoFila, ItemFila[]>(ORDEM.map((b) => [b, []]));
  const inicial = colunaInicial(colunas);
  const amanha = new Date(agora.getTime() + 24 * HORA_MS);
  const t = (iso: string) => new Date(iso).getTime();

  for (const lead of leads) {
    const tipo = tipoDaColuna(colunas, lead.coluna_id);
    if (tipo && TIPOS_FINAIS.includes(tipo)) continue;
    const add = (b: BlocoFila, motivo: string, chave: number) => grupos.get(b)!.push({ lead, motivo, chave });

    if (lead.proximo_contato_em && t(lead.proximo_contato_em) <= agora.getTime()) {
      add("atrasados", `Contato era para ${formatarDataHora(lead.proximo_contato_em)}`, t(lead.proximo_contato_em));
      continue;
    }
    if (lead.reuniao_em && tipo && TIPOS_COM_PONTOS.includes(tipo)) {
      const r = resumoPontos(calcularPontos(lead, agora));
      if (r.atrasado) {
        add("pontos_atrasados", `Atrasado: ${r.atrasado.nome}, ${formatarDataHora(r.atrasado.quando)}`, r.atrasado.quando.getTime());
        continue;
      }
    }
    if ((!lead.coluna_id || lead.coluna_id === inicial?.id) && (lead.follow_up ?? 0) === 0) {
      add("novos", `Entrou em ${formatarDataHora(lead.created_at)}`, t(lead.created_at));
      continue;
    }
    if (lead.proximo_contato_em && mesmoDiaSP(new Date(lead.proximo_contato_em), agora)) {
      add("hoje", `Contato hoje às ${formatarHora(lead.proximo_contato_em)}`, t(lead.proximo_contato_em));
      continue;
    }
    if (lead.reuniao_em && t(lead.reuniao_em) >= agora.getTime() - 2 * HORA_MS) {
      if (mesmoDiaSP(new Date(lead.reuniao_em), agora)) {
        add("reunioes_hoje", `Reunião hoje às ${formatarHora(lead.reuniao_em)}`, t(lead.reuniao_em));
        continue;
      }
      if (mesmoDiaSP(new Date(lead.reuniao_em), amanha)) {
        add("reunioes_amanha", `Reunião amanhã às ${formatarHora(lead.reuniao_em)}`, t(lead.reuniao_em));
      }
    }
  }

  return ORDEM.map((bloco) => ({
    bloco,
    titulo: TITULOS[bloco],
    itens: grupos.get(bloco)!.sort((a, b) => a.chave - b.chave),
  }));
}

// Contas do Dashboard do CRM (spec §10). Toda média vem com o tamanho da amostra. Sem dependências.
import type { ColunaCrm, EventoCrm, LeadCrm, TipoColuna } from "./crmTipos.ts";
import { FAIXAS_FATURAMENTO, ordenarColunas, rotuloFaixa, TIPOS_FINAIS } from "./crmColunas.ts";
import { ORDEM_PERIODOS, ROTULO_PERIODO, type PeriodoCall } from "./crmPontos.ts";
import { DIAS_SEMANA, HORA_MS } from "./crmFormato.ts";

export interface Amostra {
  valor: number | null;
  n: number;
}

export function media(valores: number[]): Amostra {
  return valores.length ? { valor: valores.reduce((a, b) => a + b, 0) / valores.length, n: valores.length } : { valor: null, n: 0 };
}

const horasEntre = (a: string | null | undefined, b: string | null | undefined): number | null => {
  if (!a || !b) return null;
  const h = (new Date(b).getTime() - new Date(a).getTime()) / HORA_MS;
  return Number.isFinite(h) ? Math.max(0, h) : null;
};

const soNumeros = (v: (number | null)[]) => v.filter((x): x is number => x !== null);

function tipoPor(colunas: ColunaCrm[]) {
  const m = new Map(colunas.map((c) => [c.id, c.tipo]));
  return (l: LeadCrm): TipoColuna | null => (l.coluna_id ? m.get(l.coluna_id) ?? null : null);
}

const razao = (a: number, b: number) => (b > 0 ? a / b : null);

export interface Numeros {
  leads: number;
  reunioesMarcadas: number;
  reunioesRealizadas: number;
  comparecimento: number | null;
  propostas: number;
  fechamentos: number;
  receita: number;
  ticketMedio: number | null;
  conversao: number | null;
  desqualificados: number;
}

export function calcularNumeros(leads: LeadCrm[], colunas: ColunaCrm[]): Numeros {
  const tipo = tipoPor(colunas);
  const realizadas = leads.filter((l) => l.compareceu).length;
  const faltaramSemComparecer = leads.filter((l) => (l.no_shows ?? 0) > 0 && !l.compareceu).length;
  const ganhos = leads.filter((l) => tipo(l) === "ganho");
  const receita = ganhos.reduce((s, l) => s + (Number(l.valor_fechamento) || 0), 0);
  const desqualificados = leads.filter((l) => tipo(l) === "desqualificado").length;
  return {
    leads: leads.length,
    reunioesMarcadas: leads.filter((l) => l.primeira_reuniao_marcada_em).length,
    reunioesRealizadas: realizadas,
    comparecimento: razao(realizadas, realizadas + faltaramSemComparecer),
    propostas: leads.filter((l) => l.proposta_enviada_em).length,
    fechamentos: ganhos.length,
    receita,
    ticketMedio: razao(receita, ganhos.length),
    conversao: razao(ganhos.length, leads.length - desqualificados),
    desqualificados,
  };
}

export interface Jornada {
  ateContato: Amostra;
  contatoAteReuniao: Amostra;
  followUpsAteReuniao: Amostra;
  realizadaAteProposta: Amostra;
  contatoAteFechamento: Amostra;
  followUpsAteFechamento: Amostra;
}

/** Tempos em horas; follow-ups em quantidade. */
export function calcularJornada(leads: LeadCrm[]): Jornada {
  const de = (f: (l: LeadCrm) => number | null) => media(soNumeros(leads.map(f)));
  return {
    ateContato: de((l) => horasEntre(l.created_at, l.primeiro_contato_em)),
    contatoAteReuniao: de((l) => horasEntre(l.primeiro_contato_em, l.primeira_reuniao_marcada_em)),
    followUpsAteReuniao: de((l) => (l.primeira_reuniao_marcada_em ? l.follow_ups_ate_reuniao ?? 0 : null)),
    realizadaAteProposta: de((l) => horasEntre(l.primeira_reuniao_realizada_em, l.proposta_enviada_em)),
    contatoAteFechamento: de((l) => (l.fechado_em ? horasEntre(l.primeiro_contato_em ?? l.created_at, l.fechado_em) : null)),
    followUpsAteFechamento: de((l) => (l.fechado_em ? l.follow_ups_ate_fechamento ?? 0 : null)),
  };
}

export interface EtapaFunil {
  id: string;
  rotulo: string;
  qtd: number;
  /** Passagem da etapa anterior para esta */
  taxa: number | null;
}

const ETAPAS_FUNIL: [string, string, (l: LeadCrm) => boolean][] = [
  ["leads", "Leads", () => true],
  ["contatados", "Contatados", (l) => !!l.primeiro_contato_em],
  ["marcadas", "Reunião marcada", (l) => !!l.primeira_reuniao_marcada_em],
  ["realizadas", "Reunião realizada", (l) => !!l.primeira_reuniao_realizada_em || l.compareceu],
  ["propostas", "Proposta", (l) => !!l.proposta_enviada_em],
  ["fechados", "Fechado", (l) => !!l.fechado_em],
];

export function calcularFunil(leads: LeadCrm[]): { etapas: EtapaFunil[]; gargalo: string | null } {
  const etapas: EtapaFunil[] = [];
  for (const [id, rotulo, passou] of ETAPAS_FUNIL) {
    const qtd = leads.filter(passou).length;
    const anterior = etapas[etapas.length - 1];
    etapas.push({ id, rotulo, qtd, taxa: anterior ? razao(qtd, anterior.qtd) : null });
  }
  let gargalo: string | null = null;
  let pior = Infinity;
  for (const e of etapas) {
    if (e.taxa !== null && e.taxa < pior) {
      pior = e.taxa;
      gargalo = e.id;
    }
  }
  return { etapas, gargalo };
}

export function tentativasAteReuniao(leads: LeadCrm[], colunas: ColunaCrm[]) {
  const tipo = tipoPor(colunas);
  const marcaram = leads.filter((l) => l.primeira_reuniao_marcada_em).map((l) => l.follow_ups_ate_reuniao ?? 0);
  const max = Math.max(7, ...marcaram);
  const distribuicao = Array.from({ length: max + 1 }, (_, n) => ({ n, qtd: marcaram.filter((x) => x === n).length }));
  const agora = new Map<number, number>();
  for (const l of leads) if (tipo(l) === "follow_up") agora.set(l.follow_up ?? 0, (agora.get(l.follow_up ?? 0) ?? 0) + 1);
  return {
    distribuicao,
    mediaQuemPrecisou: media(marcaram.filter((x) => x > 0)),
    emFollowUp: [...agora.entries()].sort((a, b) => a[0] - b[0]).map(([n, qtd]) => ({ n, qtd })),
  };
}

export interface LinhaGrupo {
  grupo: string;
  leads: number;
  reunioes: number;
  realizadas: number;
  fechamentos: number;
  receita: number;
  conversao: number | null;
  ticket: number | null;
}

export function porGrupo(leads: LeadCrm[], colunas: ColunaCrm[], chave: (l: LeadCrm) => string): LinhaGrupo[] {
  const tipo = tipoPor(colunas);
  const grupos = new Map<string, LeadCrm[]>();
  for (const l of leads) {
    const g = chave(l);
    grupos.set(g, [...(grupos.get(g) ?? []), l]);
  }
  return [...grupos.entries()]
    .map(([grupo, ls]) => {
      const ganhos = ls.filter((l) => tipo(l) === "ganho");
      const receita = ganhos.reduce((s, l) => s + (Number(l.valor_fechamento) || 0), 0);
      const desq = ls.filter((l) => tipo(l) === "desqualificado").length;
      return {
        grupo,
        leads: ls.length,
        reunioes: ls.filter((l) => l.primeira_reuniao_marcada_em).length,
        realizadas: ls.filter((l) => l.compareceu).length,
        fechamentos: ganhos.length,
        receita,
        conversao: razao(ganhos.length, ls.length - desq),
        ticket: razao(receita, ganhos.length),
      };
    })
    .sort((a, b) => b.leads - a.leads || a.grupo.localeCompare(b.grupo));
}

export const chaveTrafego = (l: LeadCrm) => (l.trafego === "pago" ? "Pago" : "Orgânico");
export const chaveUtmSource = (l: LeadCrm) => l.utm_source?.trim() || "Sem UTM";
export const chaveUtmCampaign = (l: LeadCrm) => l.utm_campaign?.trim() || "Sem UTM";
export const chaveNicho = (l: LeadCrm) => l.nicho?.trim() || "Sem nicho";
export const chaveFaturamento = (l: LeadCrm) => rotuloFaixa(FAIXAS_FATURAMENTO, l.faturamento_faixa);

export function contarMotivos(leads: LeadCrm[], colunas: ColunaCrm[], tipoAlvo: "perdido" | "desqualificado") {
  const tipo = tipoPor(colunas);
  const cont = new Map<string, number>();
  for (const l of leads) {
    if (tipo(l) !== tipoAlvo) continue;
    const m = (tipoAlvo === "perdido" ? l.motivo_perda : l.motivo_desqualificacao)?.trim() || "Motivo não registrado";
    cont.set(m, (cont.get(m) ?? 0) + 1);
  }

export function comparecimentoPor(eventos: EventoCrm[], chave: "periodo" | "dia_semana" | "pontos"): LinhaComparecimento[] {
  const cont = new Map<number, { grupo: string; realizadas: number; faltas: number }>();
  for (const e of eventos) {
    if (e.tipo !== "reuniao_resultado" || !e.dados?.reuniao_em) continue;
    let ordem: number;
    let grupo: string;
    if (chave === "periodo") {
      const p = e.dados.periodo as PeriodoCall;
      ordem = ORDEM_PERIODOS.indexOf(p);
      if (ordem < 0) continue;
      grupo = ROTULO_PERIODO[p];
    } else if (chave === "dia_semana") {
      const d = Number(e.dados.dia_semana);
      if (!(d >= 0 && d <= 6)) continue;
      ordem = (d + 6) % 7; // segunda primeiro
      grupo = DIAS_SEMANA[d];
    } else {
      const n = Number(e.dados.pontos_feitos);
      if (!(n >= 0)) continue;
      ordem = n;
      grupo = `${n} ${n === 1 ? "ponto" : "pontos"}`;
    }
    const linha = cont.get(ordem) ?? { grupo, realizadas: 0, faltas: 0 };
    if (e.dados.compareceu) linha.realizadas++;
    else linha.faltas++;
    cont.set(ordem, linha);
  }
  return [...cont.entries()]
    .sort((a, b) => a[0] - b[0])
    .map(([, l]) => ({ ...l, taxa: razao(l.realizadas, l.realizadas + l.faltas) }));
}

export function tempoPorColuna(leads: LeadCrm[], eventos: EventoCrm[], colunas: ColunaCrm[], agora: Date): TempoColuna[] {
  const tipo = tipoPor(colunas);
  const porLead = new Map<string, EventoCrm[]>();
  for (const e of eventos) {
    if (e.tipo !== "mudanca_coluna") continue;
    porLead.set(e.lead_id, [...(porLead.get(e.lead_id) ?? []), e]);
  }
  const duracoes = new Map<string, number[]>();
  const somar = (col: string | null | undefined, h: number | null) => {
    if (!col || h === null) return;
    duracoes.set(col, [...(duracoes.get(col) ?? []), h]);
  };
  for (const l of leads) {
    const evs = (porLead.get(l.id) ?? []).sort((a, b) => a.created_at.localeCompare(b.created_at));
    let inicio = l.created_at;
    for (const e of evs) {
      somar(e.dados.de as string | null, horasEntre(inicio, e.created_at));
      inicio = e.created_at;
    }
    const tp = tipo(l);
    if (l.coluna_id && tp && !TIPOS_FINAIS.includes(tp)) somar(l.coluna_id, horasEntre(l.coluna_desde ?? inicio, agora.toISOString()));
  }
  return ordenarColunas(colunas)
    .filter((c) => !TIPOS_FINAIS.includes(c.tipo))
    .map((c) => ({ colunaId: c.id, titulo: c.titulo, media: media(duracoes.get(c.id) ?? []) }));
}

export function previsao(leads: LeadCrm[], colunas: ColunaCrm[]) {
  const tipo = tipoPor(colunas);
  const emProposta = leads.filter((l) => tipo(l) === "proposta" || tipo(l) === "follow_up_proposta").length;
  const comDesfecho = leads.filter((l) => {
    const tp = tipo(l);
    return !!l.proposta_enviada_em && !!tp && TIPOS_FINAIS.includes(tp);
  });
  const ganhos = comDesfecho.filter((l) => tipo(l) === "ganho");
  const taxa = razao(ganhos.length, comDesfecho.length);
  const { ticketMedio } = calcularNumeros(leads, colunas);
  return {
    emProposta,
    taxa,
    ticket: ticketMedio,
    valor: taxa !== null && ticketMedio !== null ? emProposta * taxa * ticketMedio : null,
    amostra: comDesfecho.length,
  };
}

 * Executa a ação no Google Agenda. Nunca lança: em caso de falha devolve evento_status "erro" e a mensagem,
 * para o movimento continuar valendo e o card oferecer "tentar de novo".
 */
export async function executarAgenda(
  antes: LeadCrm,
  depois: LeadCrm,
  acao: AcaoAgenda,
): Promise<{ patch: Partial<LeadCrm>; evento: NovoEvento | null }> {
  if (!acao) return { patch: {}, evento: null };
  const criar = async () => {
    const ev = await criarEvento(AGENDA, corpoNovoEvento(depois, crypto.randomUUID()) as unknown as EventoGoogle, enviaConvite(depois));
    return {
      patch: { evento_google_id: ev.id ?? null, evento_meet_link: linkMeetDoEvento(ev as any), evento_status: "ok", evento_erro: null },
      evento: {
        tipo: "agenda" as const,
        descricao: enviaConvite(depois)
          ? `Evento criado na agenda com Meet; convite enviado para ${depois.email}`
          : "Evento criado na agenda com Meet (sem convidado)",
        dados: { acao: "criar", evento_id: ev.id },
      },
    };
  };
  try {
    if (acao === "criar" || !antes.evento_google_id) return await criar();
    const id = antes.evento_google_id;
    if (acao === "cancelar") {
      await excluirEvento(AGENDA, id, enviaConvite(antes)).catch((e) => {
        if (!(e instanceof AgendaEventoSumiuError)) throw e;
      });
      return {
        patch: { evento_google_id: null, evento_meet_link: null, evento_status: null, evento_erro: null },
        evento: { tipo: "agenda", descricao: "Evento cancelado na agenda", dados: { acao, evento_id: id } },
      };
    }
    let atual: EventoGoogle;
    try {
      atual = await obterEvento(AGENDA, id);
    } catch (e) {
      if (e instanceof AgendaEventoSumiuError) return await criar();
      throw e;
    }
    if ((atual as any).status === "cancelled") return await criar();
    const convidados = ((atual as any).attendees ?? []) as { email?: string; self?: boolean }[];
    const corpo = acao === "mover" ? corpoMoverEvento(depois, convidados, antes.email) : corpoConvidar(depois, convidados, antes.email);
    const enviar = (corpo.attendees?.length ?? 0) > 0 || enviaConvite(antes);
    const ev = await atualizarEvento(AGENDA, id, corpo as unknown as EventoGoogle, enviar);
    return {
      patch: { evento_meet_link: linkMeetDoEvento(ev as any) ?? depois.evento_meet_link, evento_status: "ok", evento_erro: null },
      evento: {
        tipo: "agenda",
        descricao: acao === "mover" ? "Evento movido na agenda" : `Convite enviado para ${depois.email}`,
        dados: { acao, evento_id: id },
      },
    };
  } catch (e: any) {
    const msg = e?.message ?? String(e);
    return {
      patch: { evento_status: "erro", evento_erro: msg },
      evento: { tipo: "agenda", descricao: `Falha na agenda: ${msg}`, dados: { acao, erro: msg } },
    };
  }
}

      await atualizarLead(lead.id, r.patch);
      final = { ...depois, ...r.patch };
    }
    if (r.evento) eventos.push(r.evento);
  }
  const gravados = await gravarEventos(lead.id, eventos);
  return { lead: final, eventos: gravados };
}

  leadsRef.current = leads;

  const recarregar = useCallback(async () => {
    try {
      const [c, l, e] = await Promise.all([dados.carregarColunas(), dados.carregarLeads(), dados.carregarEventos()]);
      setColunas(c);
      setLeads(l);
      setEventos(e);
      setErro(null);
    } catch (e: any) {
      setErro(e?.message ?? String(e));
    } finally {
      setCarregando(false);
    }
  }, [setColunas, setLeads, setEventos]);

  useEffect(() => {
    void recarregar();
  }, [recarregar]);

  const colunas = useMemo(() => ordenarColunas(colunasBrutas), [colunasBrutas]);
  const colunaDoLead = useCallback(
    (l: LeadCrm): ColunaCrm => colunas.find((c) => c.id === l.coluna_id) ?? colunaInicial(colunas) ?? colunas[0],
    [colunas],
  );

  const trocarLead = useCallback((l: LeadCrm) => setLeads((prev) => prev.map((x) => (x.id === l.id ? l : x))), [setLeads]);
  const juntarEventos = useCallback((novos: EventoCrm[]) => setEventos((prev) => [...novos, ...prev]), [setEventos]);

  /** Aplica o plano com atualização otimista; `base` é o estado de antes (para desfazer se falhar). */
  const executarPlano = useCallback(
    async (base: LeadCrm, plano: PlanoMovimento): Promise<LeadCrm | null> => {
      trocarLead({ ...base, ...plano.patch } as LeadCrm);
      try {
        const r = await dados.aplicarPlano(base, plano);
        trocarLead(r.lead);
        juntarEventos(r.eventos);
        if (plano.agenda && r.lead.evento_status === "erro") {
          toast.error(`Agenda: ${r.lead.evento_erro ?? "falhou"}. O lead foi salvo; use "Tentar de novo" no card.`);
        }
        return r.lead;
      } catch (e: any) {
        trocarLead(base);
        toast.error("Não foi possível salvar: " + (e?.message ?? String(e)));
        return null;
      }
    },
    [trocarLead, juntarEventos],
  );

  const moverLead = useCallback(
    async (lead: LeadCrm, destinoId: string, dadosJanela: DadosJanela) => {
      const destino = colunas.find((c) => c.id === destinoId);
      if (!destino) return false;
      const origem = colunaDoLead(lead);
      if (origem.id === destino.id) return true;
      const erroV = validarMovimento(lead, destino, dadosJanela);
      if (erroV) {
        toast.error(erroV);
        return false;
      }
      const r = await executarPlano(lead, planejarMovimento(lead, origem, destino, dadosJanela, new Date()));
      if (r && destino.tipo === "ganho" && !r.cliente_id) {
        toast.success("Lead fechado!", { action: { label: "Cadastrar como cliente", onClick: () => setCadastroCliente(r) } });
      }
      return !!r;
    },
    [colunas, colunaDoLead, executarPlano],
  );

  const pedirMovimento = useCallback(
    (lead: LeadCrm, destinoId: string, opts?: { aoCancelar?: () => void }) => {
      const destino = colunas.find((c) => c.id === destinoId);
      if (!destino || destino.id === colunaDoLead(lead).id) {
        opts?.aoCancelar?.();
        return;
      }
      const exigencia = exigenciaDe(destino.tipo, lead);
      if (exigencia === "nenhuma") {
        void moverLead(lead, destinoId, {}).then((ok) => {
          if (!ok) opts?.aoCancelar?.();
        });
        return;
      }
      setMovimentoPendente({ lead, destino, exigencia, aoCancelar: opts?.aoCancelar });
    },
    [colunas, colunaDoLead, moverLead],
  );

  const resolverMovimento = useCallback(
    async (dadosJanela: DadosJanela | null) => {
      const p = movimentoPendente;
      if (!p) return true;
      if (!dadosJanela) {
        setMovimentoPendente(null);
        p.aoCancelar?.();
        return true;
      }
      const erroV = validarMovimento(p.lead, p.destino, dadosJanela);
      if (erroV) {
        toast.error(erroV);
        return false;
      }
      setMovimentoPendente(null);
      const ok = await moverLead(p.lead, p.destino.id, dadosJanela);
      if (!ok) p.aoCancelar?.();
      return true;
    },
    [movimentoPendente, moverLead],
  );

  const registrarTentativa = useCallback(
    async (lead: LeadCrm) => {
      const r = planejarTentativa(lead, colunaDoLead(lead), new Date());
      if (typeof r === "string") {
        toast.error(r);
        return;
      }
      await executarPlano(lead, vazio({ patch: r.patch, eventos: r.eventos }));
    },
    [colunaDoLead, executarPlano],
  );

  const marcarPonto = useCallback(
    async (lead: LeadCrm, n: number, feito: boolean) => {
      if (!lead.reuniao_em) return;
      const estado = planejarPonto(lead, n, feito, new Date());
      const evento: NovoEvento = {
        tipo: "ponto_contato",
        descricao: `${feito ? "Ponto de contato feito" : "Ponto de contato desmarcado"}: ${NOMES_PONTOS[n - 1]}`,
        dados: { n, feito },
      };
      await executarPlano(lead, vazio({ patch: { pontos_contato: estado }, eventos: [evento] }));
    },
    [executarPlano],
  );

  const salvarCampos = useCallback(
    async (lead: LeadCrm, patch: Partial<LeadCrm>, rotulo?: string) => {
      const mudou = Object.fromEntries(
        Object.entries(patch).filter(([k, v]) => (lead as any)[k] !== v),
      ) as Partial<LeadCrm>;
      if (!Object.keys(mudou).length) return;
      const reuniaoFutura = !!lead.reuniao_em && new Date(lead.reuniao_em).getTime() > Date.now();
      const agenda = "email" in mudou && lead.evento_google_id && reuniaoFutura ? "convidar" : null;
      const evento: NovoEvento = { tipo: "edicao", descricao: rotulo ?? `Dados editados: ${Object.keys(mudou).join(", ")}`, dados: { campos: Object.keys(mudou) } };
      await executarPlano(lead, vazio({ patch: mudou, eventos: [evento], agenda }));
    },
    [executarPlano],
  );

  const alterarReuniao = useCallback(
    async (lead: LeadCrm, r: DadosReuniao) => {
      const erroV = validarReuniao(r);
      if (erroV) {
        toast.error(erroV);
        return false;
      }
      return !!(await executarPlano(lead, planejarReuniao(lead, r, new Date())));
    },
    [executarPlano],
  );

  const tentarAgendaDeNovo = useCallback(
    async (lead: LeadCrm) => {
      const acao = acaoPendente(lead, colunaDoLead(lead).tipo);
      if (!acao) {
        toast.info("Nada pendente na agenda para este lead.");
        return;
      }
      const r = await executarPlano(lead, vazio({ agenda: acao }));
      if (r?.evento_status === "ok") toast.success("Agenda atualizada.");
    },
    [colunaDoLead, executarPlano],
  );

  const adicionarNota = useCallback(
    async (lead: LeadCrm, texto: string) => {
      const t = texto.trim();
      if (!t) return;
      try {
        juntarEventos(await dados.gravarEventos(lead.id, [{ tipo: "nota", descricao: t }]));
      } catch (e: any) {
        toast.error("Não foi possível salvar a nota: " + (e?.message ?? e));
      }
    },
    [juntarEventos],
  );

  const criarLead = useCallback(
    async (dadosLead: Partial<LeadCrm>) => {
      const inicial = colunaInicial(colunas);
      if (!inicial) {
        toast.error("Crie uma coluna antes de cadastrar leads.");
        return null;
      }
      try {
        const r = await dados.criarLead(dadosLead, inicial);
        setLeads((prev) => [r.lead, ...prev]);
        juntarEventos(r.eventos);
        return r.lead;
      } catch (e: any) {
        toast.error("Não foi possível cadastrar: " + (e?.message ?? e));
        return null;
      }
    },
    [colunas, setLeads, juntarEventos],
  );

  const excluirLeads = useCallback(
    async (ids: string[]) => {
      try {
        await dados.excluirLeads(ids);
        const fora = new Set(ids);
        setLeads((prev) => prev.filter((l) => !fora.has(l.id)));
        setEventos((prev) => prev.filter((e) => !fora.has(e.lead_id)));
        if (fichaId && fora.has(fichaId)) setFichaId(null);
        toast.success(ids.length === 1 ? "Lead excluído." : `${ids.length} leads excluídos.`);
      } catch (e: any) {
        toast.error("Não foi possível excluir: " + (e?.message ?? e));
      }
    },
    [setLeads, setEventos, fichaId],
  );

  const moverEmMassa = useCallback(
    async (ids: string[], destinoId: string, motivo?: string) => {
      const destino = colunas.find((c) => c.id === destinoId);
      if (!destino) return;
      const modo = emMassa(destino.tipo);
      if (modo === "nao") {
        toast.error(`"${destino.titulo}" pede um dado por lead: mova um de cada vez.`);
        return;
      }
      if (modo === "motivo" && !motivo?.trim()) {
        toast.error("Escolha o motivo.");
        return;
      }
      let movidos = 0;
      for (const id of ids) {
        const lead = leadsRef.current.find((l) => l.id === id);
        if (!lead || colunaDoLead(lead).id === destino.id) continue;
        const plano = planejarMovimento(lead, colunaDoLead(lead), destino, modo === "motivo" ? { motivo } : {}, new Date());
        plano.eventos.push({ tipo: "em_massa", descricao: `Movido em massa (${ids.length} leads)`, dados: { destino: destino.id } });
        if (await executarPlano(lead, plano)) movidos++;
      }
      toast.success(`${movidos} ${movidos === 1 ? "lead movido" : "leads movidos"} para ${destino.titulo}.`);
    },
    [colunas, colunaDoLead, executarPlano],
  );

  const trafegoEmMassa = useCallback(
    async (ids: string[], trafego: Trafego) => {
      for (const id of ids) {
        const lead = leadsRef.current.find((l) => l.id === id);
        if (!lead || lead.trafego === trafego) continue;
        await executarPlano(lead, vazio({
          patch: { trafego },
          eventos: [{ tipo: "em_massa", descricao: `Etiqueta alterada em massa para ${trafego === "pago" ? "Pago" : "Orgânico"}`, dados: { trafego } }],
        }));
      }
      toast.success("Etiqueta atualizada.");
    },
    [executarPlano],
  );

  const salvarColunas = useCallback(
    async (novas: ColunaCrm[], remocoes: { id: string; destinoId: string }[]) => {
      try {
        await dados.salvarColunas(novas, remocoes);
        await recarregar();
        toast.success("Colunas salvas.");
        return true;
      } catch (e: any) {
        toast.error("Não foi possível salvar as colunas: " + (e?.message ?? e));
        return false;
      }
    },
    [recarregar],
  );

  const salvarOrdem = useCallback(async (itens: { id: string; ordem: number }[]) => {
    try {
      await dados.salvarOrdem(itens);
    } catch {
      // ordem é cosmética: a próxima carga corrige
    }
  }, []);

  const concluirCadastroCliente = useCallback(
    async (lead: LeadCrm, clienteId: string) => {
      await executarPlano(lead, vazio({
        patch: { cliente_id: clienteId },
        eventos: [{ tipo: "cliente_cadastrado", descricao: "Cadastrado como cliente ativo", dados: { cliente_id: clienteId } }],
      }));
    },
    [executarPlano],
  );

  const valor: CrmValor = {
    colunas, leads, eventos, carregando, erro, setLeads, recarregar, colunaDoLead, pedirMovimento, movimentoPendente,
    resolverMovimento, moverLead, registrarTentativa, marcarPonto, salvarCampos, alterarReuniao, tentarAgendaDeNovo,
    adicionarNota, criarLead, excluirLeads, moverEmMassa, trafegoEmMassa, salvarColunas, salvarOrdem,
    fichaId, abrirFicha: setFichaId, cadastroCliente, iniciarCadastroCliente: setCadastroCliente, concluirCadastroCliente,
  };
  return <Ctx.Provider value={valor}>{children}</Ctx.Provider>;
}

function lerLocal<T>(chave: string, padrao: T): T {
  try {
    const v = typeof window !== "undefined" ? window.localStorage.getItem(chave) : null;
    return v ? { ...padrao, ...JSON.parse(v) } : padrao;
  } catch {
    return padrao;
  }
}

function gravarLocal(chave: string, valor: unknown) {
  try {
    window.localStorage.setItem(chave, JSON.stringify(valor));
  } catch {
    // navegador sem armazenamento: só não lembra a escolha
  }
}

/** Relógio da tela: urgências e pontos mudam com o tempo */
function useAgora(intervaloMs: number) {
  const [agora, setAgora] = useState(() => new Date());
  useEffect(() => {
    const t = setInterval(() => setAgora(new Date()), intervaloMs);
    return () => clearInterval(t);
  }, [intervaloMs]);
  return agora;
}
